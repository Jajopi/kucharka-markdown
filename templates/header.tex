% LaTeX preamble additions for the PDF (KOMA-Script scrreprt + XeLaTeX).
% Mimics the style of the original guidebook: Latin Modern body text,
% sans-serif headings, centred unnumbered sections, running header.

\usepackage{qrcode}
% footnote numbers at the bottom of the page link back to the place in the text
\usepackage[numberlinked=false]{footnotebackref}

% Fallback font for characters missing in Latin Modern (☺, →, π, …);
% kucharka.lua wraps such characters in \kfallback{…}.
\IfFontExistsTF{DejaVu Sans}{\newfontfamily\kfallbackfont{DejaVu Sans}}{%
  \IfFontExistsTF{Noto Sans Symbols}{\newfontfamily\kfallbackfont{Noto Sans Symbols}}{%
    \IfFontExistsTF{FreeSerif}{\newfontfamily\kfallbackfont{FreeSerif}}{}}}
\providecommand{\kfallbackfont}{}
\IfFontExistsTF{Noto Sans Math}{\newfontfamily\kfallbackmath{Noto Sans Math}}{\let\kfallbackmath\kfallbackfont}
\newcommand{\kfallback}[1]{{\kfallbackfont #1}}
\newcommand{\kfallbackm}[1]{{\kfallbackmath #1}}

% headings
\setkomafont{disposition}{\normalfont\sffamily}
\setkomafont{chapter}{\sffamily\bfseries\LARGE}
\setkomafont{section}{\sffamily\Large}
\setkomafont{subsection}{\sffamily\large}
\setkomafont{subsubsection}{\sffamily\normalsize\bfseries}
\renewcommand*{\raggedsection}{\centering}
\providecommand*{\raggedchapter}{}
\renewcommand*{\raggedchapter}{\raggedright}
\RedeclareSectionCommand[beforeskip=0pt, afterskip=1.2\baselineskip]{chapter}
\RedeclareSectionCommand[beforeskip=-1.5\baselineskip, afterskip=0.6\baselineskip]{section}

% running header "Kapitola 1. Název" with a rule, page number at the bottom
\usepackage[headsepline]{scrlayer-scrpage}
\pagestyle{scrheadings}
\clearpairofpagestyles
\automark[chapter]{chapter}
\renewcommand*{\chaptermarkformat}{\chapapp~\thechapter.\enskip}
\ohead{\headmark}
\cfoot*{\pagemark}
\setkomafont{pageheadfoot}{\normalfont\sffamily}
\setkomafont{pagenumber}{\normalfont\sffamily}
\renewcommand*{\chapterpagestyle}{scrheadings}

% table of contents: chapters bold serif, sections with dots
\setkomafont{partentry}{\normalfont\bfseries\Large}
\setkomafont{chapterentry}{\normalfont\bfseries}
\setkomafont{chapterentrypagenumber}{\normalfont\bfseries}
% entries are links in the link colour (toccolor in pdf.yaml), numbers stay black
\newcommand*{\ktocnumber}[1]{\textcolor{black}{#1}}
\DeclareTOCStyleEntry[entrynumberformat=\ktocnumber, numwidth=2.3em]{tocline}{chapter}
\DeclareTOCStyleEntry[entrynumberformat=\ktocnumber, indent=2.3em]{tocline}{section}
\DeclareTOCStyleEntry[entrynumberformat=\ktocnumber, indent=4.6em]{tocline}{subsection}

% lists: long terms in definition lists may wrap, the description starts on a
% new line (kucharka.lua inserts the break and forbids a page break there)
\usepackage{enumitem}
\setlist[description]{style=unboxed, leftmargin=2em}

% keep headings / short tables together with what follows
\usepackage{needspace}

% links and footnotes (colours used by pdf.yaml)
\definecolor{klink}{HTML}{C0392B}
\definecolor{kurl}{HTML}{2A6EBB}
\deffootnote{1.5em}{1em}{\textsuperscript{\hyperref[\BackrefFootnoteTag]{\thefootnotemark}}\,}

% the title page is generated by kucharka.lua (include-before)
\renewcommand{\maketitle}{}
